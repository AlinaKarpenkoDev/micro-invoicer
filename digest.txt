# ==============================================================================
# .semgrep-ai-rules.yml
# Комплексний набір правил для виявлення антипатернів згенерованого AI коду
#
# Категорії:
# 1. Архитектурні та структурні антипатерни (Premature Abstraction / Over-Engineering)
# 2. Типізація та контрактні дефекти (Type Evasion / Partial-Everything)
# 3. Надійність та обробка помилок (Exception Swallowing / Missing Error Returns)
# 4. Асинхронність та ресурси (Unawaited Promises / Sync in Async / Leaks)
# 5. Безпека та ізоляція (Injection Traps / Hardcoded Secrets / Permissive Defaults)
# 6. Тестування та верифікація (Tautological Tests / Implementation Testing)
# 7. Екосистемні та середовищні аномалії (Zombie Dependencies / Version Anachronism)
# 8. Ментальні антипатерни розробника (Process Smells: Vibe Coding / Whack-a-Mole)
# ==============================================================================

rules:
  # ============================================================================
  # 1. АРХІТЕКТУРНІ ТА СТРУКТУРНІ АНТИПАТЕРНИ
  # ============================================================================

  - id: ai-anti-pattern-overengineering-naming-bloat
    message: |
      AI Anti-pattern (Premature Abstraction / Over-Engineering):
      Роздута назва класу/фабрики ($CLASS).
      AI схильний генерувати 4–5 шарів абстракцій (EventDispatcherServiceFactory,
      StrategyManagerProvider тощо) для задач, що вирішуються чистою функцією на 10 рядків.
    severity: WARNING
    languages: [ts, js]
    patterns:
      - pattern: class $CLASS { ... }
      - metavariable-regex:
          metavariable: $CLASS
          regex: ".*(DispatcherServiceFactory|ManagerProviderStrategy|ServiceFactoryStrategy|EventDispatcherFactory)$"
    metadata:
      category: ai-antipattern
      subcategory: architecture
      impact: MEDIUM

  # ============================================================================
  # 2. ТИПІЗАЦІЯ ТА КОНТРАКТНІ ДЕФЕКТИ
  # ============================================================================

  - id: ai-anti-pattern-type-evasion-any
    message: |
      AI Anti-pattern (Type Evasion / "any"-ification): Використання 'as any'.
      Коли виникає конфлікт дженериків, AI обирає шлях найменшого опору, знищуючи
      гарантії компілятора. Використовуйте дженерики або звуження типів (narrowing).
    severity: ERROR
    languages: [ts]
    pattern: $X as any
    metadata:
      category: ai-antipattern
      subcategory: type-safety
      impact: HIGH

  - id: ai-anti-pattern-type-evasion-unknown-cast
    message: |
      AI Anti-pattern (Type Evasion): Подвійне приведення '$X as unknown as $T'.
      Класичний хак нейромереж для обходу несумісності типів TypeScript.
      Застосовуйте рантайм-валідацію через Zod або перегляньте дизайн інтерфейсу.
    severity: ERROR
    languages: [ts]
    pattern: $X as unknown as $T
    metadata:
      category: ai-antipattern
      subcategory: type-safety
      impact: HIGH

  - id: ai-anti-pattern-non-null-assertion
    message: |
      AI Anti-pattern (Type Evasion): Оператор '!' (non-null assertion).
      AI глушить перевірку null/undefined знаком '!', ризикуючи падінням у рантаймі.
      Використовуйте 'if (x == null)' або optional chaining '?.'.
    severity: WARNING
    languages: [ts]
    pattern-either:
      - pattern: $X!.$PROP
      - pattern: ($X)!.$PROP
    metadata:
      category: ai-antipattern
      subcategory: type-safety
      impact: MEDIUM

  - id: ai-anti-pattern-unvalidated-json-parse-cast
    message: |
      AI Anti-pattern (No Boundary Defense): 'JSON.parse(...) as $T'.
      Сліпе припущення, що зовнішні дані завжди відповідають типу $T.
      При зміні формату даних код впаде з Cannot read properties of undefined.
      Обов'язково використовуйте Zod (schema.parse(JSON.parse(...))) або TypeBox.
    severity: ERROR
    languages: [ts]
    pattern: JSON.parse(...) as $T
    metadata:
      category: ai-antipattern
      subcategory: data-validation
      impact: HIGH

  # ============================================================================
  # 3. НАДІЙНІСТЬ ТА ОБРОБКА ПОМИЛОК
  # ============================================================================

  - id: ai-anti-pattern-empty-catch
    message: |
      AI Anti-pattern (Exception Swallowing): Порожній блок 'catch'.
      AI часто ховає помилки, щоб код 'не падав', залишаючи застосунок у невалідному зомбі-стані.
      Обробіть виняток або прокиньте його далі (throw).
    severity: ERROR
    languages: [ts, js]
    pattern-either:
      - pattern: |
          try {
            ...
          } catch ($E) {
          }
      - pattern: |
          try {
            ...
          } catch {
          }
    metadata:
      category: ai-antipattern
      subcategory: error-handling
      impact: CRITICAL

  - id: ai-anti-pattern-catch-only-console
    message: |
      AI Anti-pattern (Exception Swallowing): Блок 'catch' містить лише console.log/error.
      Помилка ігнорується, і функція несподівано повертає undefined, що призводить
      до ланцюгового падіння кількома кроками пізніше.
    severity: WARNING
    languages: [ts, js]
    pattern-either:
      - pattern: |
          try {
            ...
          } catch ($E) {
            console.$LOG($...ARGS);
          }
      - pattern: |
          try {
            ...
          } catch {
            console.$LOG($...ARGS);
          }
    metadata:
      category: ai-antipattern
      subcategory: error-handling
      impact: HIGH

  - id: ai-anti-pattern-swallowed-promise-catch
    message: |
      AI Anti-pattern (Exception Swallowing): Порожній або лише логуючий .catch() у промісі.
      Приховує збій асинхронної операції. Додайте повноцінну обробку помилки або прокидання.
    severity: WARNING
    languages: [ts, js]
    pattern-either:
      - pattern: $P.catch(() => {})
      - pattern: $P.catch(($E) => {})
      - pattern: $P.catch(($E) => { console.$LOG(...); })
    metadata:
      category: ai-antipattern
      subcategory: error-handling
      impact: HIGH

  # ============================================================================
  # 4. АСИНХРОННІСТЬ ТА РЕСУРСИ
  # ============================================================================

  - id: ai-anti-pattern-async-foreach
    message: |
      AI Anti-pattern (Async Trap / Unawaited Promises): 'forEach(async (...) => { ... })'.
      Метод 'forEach' не чекає на завершення промісів. Операції виконуються неконтрольовано,
      а помилки губляться. Використовуйте 'for (const item of items)' або 'await Promise.all(items.map(...))'.
    severity: ERROR
    languages: [ts, js]
    pattern-either:
      - pattern: $ARR.forEach(async ($ITEM) => { ... })
      - pattern: $ARR.forEach(async function ($ITEM) { ... })
      - pattern: $ARR.forEach(async ($ITEM, $IDX) => { ... })
      - pattern: $ARR.forEach(async () => { ... })
    metadata:
      category: ai-antipattern
      subcategory: async-control-flow
      impact: HIGH

  - id: ai-anti-pattern-unawaited-async-map
    message: |
      AI Anti-pattern (Async Trap): Виклик '$ARR.map(async ...)' як окремий оператор без очікування.
      Масив промісів створюється, але ніколи не очікується через await Promise.all().
    severity: ERROR
    languages: [ts, js]
    patterns:
      - pattern: $ARR.map(async ($ITEM) => { ... });
      - pattern-regex: '\.map\s*\(\s*async\b'
    metadata:
      category: ai-antipattern
      subcategory: async-control-flow
      impact: HIGH

  - id: ai-anti-pattern-blocking-sync-io
    message: |
      AI Anti-pattern (Blocking I/O in Async / Sync in Async):
      Використання синхронних викликів fs.*Sync або execSync блокує Event Loop.
      У Bun використовуйте 'Bun.file(...).text()', 'Bun.write(...)', або 'node:fs/promises'.
    severity: WARNING
    languages: [ts, js]
    pattern-either:
      - pattern: fs.readFileSync(...)
      - pattern: fs.writeFileSync(...)
      - pattern: readFileSync(...)
      - pattern: writeFileSync(...)
      - pattern: child_process.execSync(...)
      - pattern: execSync(...)
    metadata:
      category: ai-antipattern
      subcategory: performance
      impact: HIGH

  # ============================================================================
  # 5. БЕЗПЕКА ТА ІЗОЛЯЦІЯ
  # ============================================================================

  - id: ai-anti-pattern-command-injection-template
    message: |
      AI Anti-pattern (Injection Trap): Інтерполяція рядків у системній команді exec(`...${...}...`).
      AI часто конкатенує змінні в системні команди, створюючи критичну вразливість Command Injection.
      Використовуйте Bun.spawn(["command", arg1, arg2]) з масивом аргументів без виклику шелу.
    severity: ERROR
    languages: [ts, js]
    pattern-either:
      - pattern: exec(`...${$VAR}...`)
      - pattern: execSync(`...${$VAR}...`)
      - pattern: child_process.exec(`...${$VAR}...`)
      - pattern: child_process.execSync(`...${$VAR}...`)
    metadata:
      category: ai-antipattern
      subcategory: security
      impact: CRITICAL

  - id: ai-anti-pattern-permissive-cors
    message: |
      AI Anti-pattern (Permissive Defaults): Встановлення надмірно відкритих прав CORS (origin: '*').
      AI часто робить це, щоб подолати локальні помилки CORS, відкриваючи доступ до API будь-якому сайту.
      Вкажіть точний домен або білий список дозволених джерел.
    severity: WARNING
    languages: [ts, js]
    pattern-either:
      - pattern: "cors({ origin: '*' })"
      - pattern: "cors({ origin: true })"
      - pattern: '$RES.setHeader("Access-Control-Allow-Origin", "*")'
    metadata:
      category: ai-antipattern
      subcategory: security
      impact: HIGH

  - id: ai-anti-pattern-permissive-tls
    message: |
      AI Anti-pattern (Permissive Defaults): Відключення TLS-верифікації (rejectUnauthorized: false).
      Небезпечний хак для усунення проблем із самопідписаними сертифікатами, що відкриває шлях до MITM-атак.
    severity: ERROR
    languages: [ts, js]
    pattern: "rejectUnauthorized: false"
    metadata:
      category: ai-antipattern
      subcategory: security
      impact: CRITICAL

  - id: ai-anti-pattern-permissive-chmod
    message: |
      AI Anti-pattern (Permissive Defaults): Встановлення прав 777 (chmod 0o777).
      AI надає повний доступ усім користувачам системи замість мінімально необхідних прав (0o755 або 0o644).
    severity: ERROR
    languages: [ts, js]
    pattern-either:
      - pattern: fs.chmodSync($PATH, 0o777)
      - pattern: fs.chmod($PATH, 0o777, ...)
      - pattern: chmodSync($PATH, 0o777)
      - pattern: chmod($PATH, 0o777, ...)
    metadata:
      category: ai-antipattern
      subcategory: security
      impact: HIGH

  # ============================================================================
  # 6. ТЕСТУВАННЯ ТА ВЕРИФІКАЦІЯ
  # ============================================================================

  - id: ai-anti-pattern-tautological-assertion
    message: |
      AI Anti-pattern (Tautological Tests): Тест перевіряє значення саме з собою або тривіальну тавтологію.
      Згенеровані тести часто 'зелені' лише тому, що нічого реального не перевіряють (expect(x).toBe(x)).
    severity: ERROR
    languages: [ts, js]
    pattern-either:
      - pattern: expect($X).toBe($X)
      - pattern: expect($X).toEqual($X)
      - pattern: expect(true).toBe(true)
      - pattern: expect(false).toBe(false)
      - pattern: assert.equal($X, $X)
      - pattern: assert.deepEqual($X, $X)
      - pattern: assert.strictEqual($X, $X)
    metadata:
      category: ai-antipattern
      subcategory: testing
      impact: HIGH

  - id: ai-anti-pattern-empty-test
    message: |
      AI Anti-pattern (Tautological / Dummy Tests): Порожнє тіло тесту без асертів.
      Тест завжди успішний, створюючи фіктивне покриття.
    severity: ERROR
    languages: [ts, js]
    pattern-either:
      - pattern: test($NAME, () => {})
      - pattern: it($NAME, () => {})
      - pattern: test($NAME, async () => {})
      - pattern: it($NAME, async () => {})
    metadata:
      category: ai-antipattern
      subcategory: testing
      impact: HIGH

  # ============================================================================
  # 7. ЕКОСИСТЕМНІ ТА СЕРЕДОВИЩНІ АНОМАЛІЇ (Zombie Dependencies & Anachronisms)
  # ============================================================================

  - id: ai-anti-pattern-zombie-lodash
    message: |
      AI Anti-pattern (Zombie Dependencies): Імпорт бібліотеки 'lodash'.
      У сучасному JavaScript більшість методів lodash замінені нативними еквівалентами:
      structuredClone(), Object.keys(), Array.prototype.flat() тощо.
    severity: WARNING
    languages: [ts, js]
    pattern-either:
      - pattern: import $X from 'lodash'
      - pattern: import $X from 'lodash-es'
      - pattern: require('lodash')
    metadata:
      category: ai-antipattern
      subcategory: zombie-dependencies
      impact: LOW

  - id: ai-anti-pattern-zombie-moment
    message: |
      AI Anti-pattern (Zombie Dependencies): Імпорт застарілої бібліотеки 'moment'.
      Moment.js офіційно перебуває в режимі підтримки і не рекомендований для нових проєктів.
      Використовуйте нативний об'єкт Intl, Date або легкі альтернативи (date-fns).
    severity: WARNING
    languages: [ts, js]
    pattern-either:
      - pattern: import $X from 'moment'
      - pattern: require('moment')
    metadata:
      category: ai-antipattern
      subcategory: zombie-dependencies
      impact: MEDIUM

  - id: ai-anti-pattern-zombie-uuid
    message: |
      AI Anti-pattern (Zombie Dependencies): Імпорт стороннього пакету 'uuid'.
      У сучасному середовищі (Bun, Node.js 18+, браузери) є вбудований нативний метод:
      'crypto.randomUUID()'. Зайва залежність не потрібна.
    severity: WARNING
    languages: [ts, js]
    pattern-either:
      - pattern: import $X from 'uuid'
      - pattern: require('uuid')
    metadata:
      category: ai-antipattern
      subcategory: zombie-dependencies
      impact: LOW

  - id: ai-anti-pattern-bun-legacy-fetch-libraries
    message: |
      AI Anti-pattern (Legacy Hallucinations): Імпорт '$LIB' у Bun проєкті.
      Bun має нативний супершвидкий глобальний fetch().
      Бібліотеки axios/node-fetch є зайвими та знижують швидкість.
    severity: WARNING
    languages: [ts, js]
    pattern-either:
      - pattern: import $X from 'axios'
      - pattern: import $X from 'node-fetch'
      - pattern: import $X from 'cross-fetch'
      - pattern: import $X from 'request'
      - pattern: import $X from 'got'
      - pattern: require('axios')
      - pattern: require('node-fetch')
      - pattern: require('cross-fetch')
    metadata:
      category: ai-antipattern
      subcategory: runtime-modernity
      impact: MEDIUM

  - id: ai-anti-pattern-bun-legacy-dotenv

# ==============================================================================
# .semgrep-engineering-rules.yml
# Класичні інженерні антипатерни в архітектурі, коді та тестах
#
# Категорії:
# 1. Архітектурні та структурні (God Object, Lasagna Code, Circular Dependencies)
# 2. Проєктування та ООП (Poltergeists, Lava Flow, Ghost Classes)
# 3. Обробка помилок і стабільність (Catch-All, Error Swallowing, Magic Numbers, Blind Trust)
# 4. Стан, асинхронність і паралелізм (Global State Pollution, Fire and Forget, Sync-in-Async)
# 5. Якість коду та Clean Code (Reinventing the Wheel, Boolean Trap, Dead Code)
# 6. Тестування (The Liar, Mock Happy, Slow Poke Tests)
# ==============================================================================

rules:
  # ============================================================================
  # 1. АРХІТЕКТУРА ТА ПРОЄКТУВАННЯ
  # ============================================================================

  - id: engineering-anti-pattern-poltergeist-class
    message: |
      Класичний антипатерн (Poltergeists / Ghost Classes):
      Клас із єдиним методом, який лише делегує виклик іншому об'єкту.
      Такі короткоживучі 'класи-привиди' роздувають архітектуру (Lasagna Code)
      без додавання цінності. Замініть клас на просту чисту функцію.
    severity: INFO
    languages: [ts, js]
    pattern: |
      class $CLASS {
        execute($...ARGS) {
          return $TARGET.$METHOD($...ARGS);
        }
      }
    metadata:
      category: engineering-antipattern
      subcategory: oop-design
      impact: LOW

  - id: engineering-anti-pattern-lava-flow-dead-branch
    message: |
      Класичний антипатерн (Lava Flow / Dead Code):
      Залишок мертвого коду в блоці 'if (false) { ... }' або 'if (0) { ... }'.
      Не залишайте закоментовані чи вимкнені гілки в кодовій базі.
      Історія змін надійно зберігається у Git — видаляйте мертвий код.
    severity: WARNING
    languages: [ts, js]
    pattern-either:
      - pattern: |
          if (false) {
            ...
          }
      - pattern: |
          if (0) {
            ...
          }
    metadata:
      category: engineering-antipattern
      subcategory: clean-code
      impact: MEDIUM

  # ============================================================================
  # 2. ОБРОБКА ПОМИЛОК І СТАБІЛЬНІСТЬ
  # ============================================================================

  - id: engineering-anti-pattern-catch-all-any
    message: |
      Класичний антипатерн (Catch-All / Error Evasion):
      Перехоплення помилки типом 'any' (catch (error: any)).
      Це маскує тип помилки і заважає компілятору контролювати властивості.
      У сучасному TypeScript використовуйте 'catch (error: unknown)'
      з явною перевіркою 'if (error instanceof Error)'.
    severity: WARNING
    languages: [ts]
    pattern: |
      try {
        ...
      } catch ($E: any) {
        ...
      }
    metadata:
      category: engineering-antipattern
      subcategory: error-handling
      impact: MEDIUM

  - id: engineering-anti-pattern-magic-timeout
    message: |
      Класичний антипатерн (Magic Numbers):
      Використання магічного числа ($NUM мс) у таймауті/інтервалі.
      Жорстко закодовані значення (86400000, 3600000 тощо) ускладнюють читання
      та конфігурування. Винесіть константу з розрахунком
      (наприклад, ONE_DAY_MS = 24 * 60 * 60 * 1000) або у файл конфігурації.
    severity: WARNING
    languages: [ts, js]
    patterns:
      - pattern-either:
          - pattern: setTimeout($CB, $NUM)
          - pattern: setInterval($CB, $NUM)
      - metavariable-regex:
          metavariable: $NUM
          regex: "^[1-9][0-9]{4,}$"
    metadata:
      category: engineering-antipattern
      subcategory: clean-code
      impact: LOW

  - id: engineering-anti-pattern-blind-trust-req-json
    message: |
      Класичний антипатерн (Blind Trust / No Boundary Defense):
      Приведення зовнішніх даних '(await req.json()) as $TYPE' без схеми валідації.
      Будь-який зовнішній клієнт може надіслати неповні або невалідні дані.
      Застосовуйте Zod (UserSchema.parse(await req.json())) для гарантії контрактів.
    severity: ERROR
    languages: [ts]
    pattern-either:
      - pattern: (await $REQ.json()) as $TYPE
      - pattern: (await $REQ.json()) as unknown as $TYPE
      - pattern: $REQ.body as $TYPE
    metadata:
      category: engineering-antipattern
      subcategory: security
      impact: HIGH

  # ============================================================================
  # 3. СТАН, АСИНХРОННІСТЬ І ПАРАЛЕЛІЗМ
  # ============================================================================

  - id: engineering-anti-pattern-global-mutable-state
    message: |
      Класичний антипатерн (Global State Pollution):
      Експорт мутабельної змінної 'export let $VAR'.
      Спільний глобальний стан призводить до непередбачуваних гонок (Race Conditions),
      робить модулі неізольованими і ламає паралельний запуск тестів.
      Використовуйте інкапсуляцію в класи/функції або Dependency Injection.
    severity: ERROR
    languages: [ts, js]
    pattern: export let $VAR = $VAL
    metadata:
      category: engineering-antipattern
      subcategory: state-management
      impact: HIGH

  - id: engineering-anti-pattern-floating-fetch
    message: |
      Класичний антипатерн (Fire and Forget / Floating Promise):
      Виклик fetch(...) як окремого виразу без await, return або обробки .catch().
      У разі мережевого збою виникне unhandledRejection, а потік виконання продовжиться
      без очікування результату.
    severity: WARNING
    languages: [ts, js]
    patterns:
      - pattern: fetch(...)
      - pattern-not-inside: await fetch(...)
      - pattern-not-inside: return fetch(...)
      - pattern-not-inside: const $X = fetch(...)
      - pattern-not-inside: let $X = fetch(...)
      - pattern-not-inside: fetch(...).then(...)
      - pattern-not-inside: fetch(...).catch(...)
    metadata:
      category: engineering-antipattern
      subcategory: async
      impact: HIGH

  # ============================================================================
  # 4. ЯКІСТЬ КОДУ ТА ЧИСТИЙ КОД (Clean Code)
  # ============================================================================

  - id: engineering-anti-pattern-boolean-trap
    message: |
      Класичний антипатерн (Boolean Trap):
      Виклик функції з кількома послідовними булевими літералами (..., true, false, ...).
      Під час читання виклику неможливо зрозуміти, за що відповідає кожен прапорець.
      Використовуйте Options Object (наприклад, { verbose: true, dryRun: false }).
    severity: WARNING
    languages: [ts, js]
    pattern-either:
      - pattern: $FN(..., true, false, ...)
      - pattern: $FN(..., false, true, ...)
      - pattern: $FN(..., true, true, ...)
      - pattern: $FN(..., false, false, ...)
    metadata:
      category: engineering-antipattern
      subcategory: clean-code
      impact: LOW

  - id: engineering-anti-pattern-reinventing-clone
    message: |
      Класичний антипатерн (Reinventing the Wheel):
      Використання 'JSON.parse(JSON.stringify($X))' для глибокого клонування об'єкта.
      Цей метод повільний, втрачає типи Date, Map, Set, RegExp та викликає помилку
      на циклічних посиланнях. Використовуйте нативний метод 'structuredClone($X)'.
    severity: WARNING
    languages: [ts, js]
    pattern: JSON.parse(JSON.stringify($X))
    metadata:
      category: engineering-antipattern
      subcategory: clean-code
      impact: MEDIUM

  # ============================================================================
  # 5. ТЕСТУВАННЯ (Testing Smells)
  # ============================================================================

  - id: engineering-anti-pattern-slow-poke-test-sleep
    message: |
      Класичний антипатерн (Slow Poke Tests / Arbitrary Sleep):
      Використання паузи sleep() або setTimeout() всередині тестів.
      Штучні затримки уповільнюють виконання тестів на CI та роблять їх нестабільними (flaky).
      Використовуйте підписку на події, polling очікування умов або мокінг системного таймера.
    severity: WARNING
    languages: [ts, js]
    patterns:
      - pattern-either:
          - pattern: await sleep(...)
          - pattern: sleep(...)
          - pattern: await Bun.sleep(...)
          - pattern: Bun.sleep(...)
      - pattern-inside: |
          test(..., ...)
    metadata:
      category: engineering-antipattern
      subcategory: testing
      impact: MEDIUM

  - id: engineering-anti-pattern-mock-happy-tautology
    message: |
      Класичний антипатерн (Mock Happy / The Liar):
      Тест одразу звіряє значення, яке сам щойно захардкодив у мок: expect($MOCK()).toBe($VAL).
      Тестується лише працездатність самого мока, а не цільова поведінка бізнес-логіки.
    severity: WARNING
    languages: [ts, js]
    pattern: |
      $MOCK.mockReturnValue($VAL);
      ...
      expect($MOCK()).toBe($VAL);
    metadata:
      category: engineering-antipattern
      subcategory: testing
      impact: HIGH

  # ============================================================================
  # 6. БАЗА ДАНИХ ТА ЧИТАБЕЛЬНІСТЬ (Database & Readability)
  # ============================================================================

  - id: engineering-anti-pattern-double-negative-boolean
    message: |
      Класичний антипатерн (Double Negative Boolean):
      Подвійне заперечення (!isNot... або !hasNo...).
      Це створює зайве ментальне навантаження. Використовуйте позитивні назви (isEnabled, hasAccess).
    severity: WARNING
    languages: [ts, js]
    pattern-either:
      - pattern: "!$OBJ.isNot$NAME"
      - pattern: "!$OBJ.hasNo$NAME"
      - pattern: "!isNot$NAME"
      - pattern: "!hasNo$NAME"
    metadata:
      category: engineering-antipattern
      subcategory: readability
      impact: LOW

  - id: engineering-anti-pattern-select-star
    message: |
      Класичний антипатерн (SELECT * / Overhead Query):
      Запит витягує всі стовпчики (SELECT *).
      Це призводить до передачі надлишкових даних, блокує оптимізацію індексів (Covering Indexes)
      та може призвести до витоку конфіденційних полів. Явно вказуйте необхідні колонки.
    severity: WARNING
    languages: [ts, js]
    pattern-regex: "(?i)\\bSELECT\\s+\\*\\s+FROM\\b"
    metadata:
      category: engineering-antipattern
      subcategory: database
      impact: MEDIUM

  - id: engineering-anti-pattern-query-in-loop
    message: |
      Класичний антипатерн (N+1 Query / Query in Loop):
      Виклик запиту до бази даних усередині циклу.
      Це генерує сотні зайвих мережевих кругообігів. Використовуйте batch-запит з оператором IN(...)
      або JOIN.
    severity: WARNING
    languages: [ts, js]
    patterns:
      - pattern-either:
          - pattern: await $DB.select(...)
          - pattern: await $DB.query(...)
          - pattern: await $DB.execute(...)
      - pattern-inside: |
          for (...) {
            ...
          }
    metadata:
      category: engineering-antipattern
      subcategory: database
      impact: HIGH

FROM node:22-alpine

RUN apk add --no-cache git openssh-client && \
    mkdir -p -m 0700 ~/.ssh && ssh-keyscan github.com >> ~/.ssh/known_hosts

WORKDIR /app

COPY package.json package-lock.json turbo.json ./
COPY apps/api/package.json ./apps/api/
COPY apps/web/package.json ./apps/web/
COPY packages/shared/package.json ./packages/shared/

RUN --mount=type=secret,id=ssh_key \
    --mount=type=secret,id=npm_token \
    eval $(ssh-agent -s) && \
    [ -f /run/secrets/ssh_key ] && ssh-add /run/secrets/ssh_key || true && \
    [ -f /run/secrets/npm_token ] && echo "//registry.npmjs.org/:_authToken=$(cat /run/secrets/npm_token)" > ~/.npmrc || true && \
    npm ci && \
    rm -f ~/.npmrc

COPY . .

RUN npx turbo run build --filter=api

RUN npm prune --omit=dev

ENV NODE_ENV=production
EXPOSE 4000

CMD sh -c "cd apps/api && npm run start:prod"

import {
  CanActivate,
  ExecutionContext,
  Injectable,
  UnauthorizedException,
} from '@nestjs/common';
import { JwtService } from '@nestjs/jwt';
import { Request } from 'express';

@Injectable()
export class AuthGuard implements CanActivate {
  constructor(private jwtService: JwtService) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const request = context
      .switchToHttp()
      .getRequest<Request & { user?: unknown }>();

    const token = this.extractTokenFromHeader(request);

    if (!token) {
      throw new UnauthorizedException('No access (token missing)');
    }

    try {
      const payload = await this.jwtService.verifyAsync<{
        sub: string;
        email: string;
      }>(token);
      request.user = payload;
    } catch {
      throw new UnauthorizedException('The token is invalid');
    }

    return true;
  }

  private extractTokenFromHeader(request: Request): string | undefined {
    const [type, token] = request.headers.authorization?.split(' ') ?? [];
    return type === 'Bearer' ? token : undefined;
  }
}

import {
  Inject,
  Injectable,
  ConflictException,
  UnauthorizedException,
  NotFoundException,
} from '@nestjs/common';
import { JwtService } from '@nestjs/jwt';
import { Kysely } from 'kysely';
import { Database } from '../database/database';
import * as bcrypt from 'bcryptjs';

@Injectable()
export class AuthService {
  constructor(
    @Inject('DB_INSTANCE') private readonly db: Kysely<Database>,
    private readonly jwtService: JwtService,
  ) {}

  async register(email: string, plainPassword: string) {
    const hashedPassword = await bcrypt.hash(plainPassword, 10);

    try {
      const newUser = await this.db
        .insertInto('users')
        .values({
          email: email,
          password_hash: hashedPassword,
          role: 'OWNER',
        })
        .returning(['id', 'email', 'created_at'])
        .executeTakeFirstOrThrow();

      await this.db
        .insertInto('workspaces')
        .values({
          user_id: newUser.id,
          name: 'Personal Workspace',
        })
        .execute();

      return newUser;
    } catch (error: unknown) {
      const pgError = error as { code?: string };
      if (pgError.code === '23505') {
        throw new ConflictException(
          'A user with this email address already exists!',
        );
      }
      throw error;
    }
  }
  async login(email: string, plainPassword: string) {
    const user = await this.db
      .selectFrom('users')
      .selectAll()
      .where('email', '=', email)
      .executeTakeFirst();

    if (!user) {
      throw new UnauthorizedException('Incorrect email address or password');
    }

    const isPasswordValid = await bcrypt.compare(
      plainPassword,
      user.password_hash,
    );

    if (!isPasswordValid) {
      throw new UnauthorizedException('Incorrect email address or password');
    }

    const payload = {
      sub: user.id,
      email: user.email,
      role: user.role,
    };

    return {
      access_token: this.jwtService.sign(payload),
    };
  }

  async impersonateUser(adminId: string, targetUserId: string) {
    const user = await this.db
      .selectFrom('users')
      .select(['id', 'email', 'role'])
      .where('id', '=', targetUserId)
      .executeTakeFirst();

    if (!user) {
      throw new NotFoundException('Page not found');
    }

    const payload = {
      sub: user.id,
      email: user.email,
      role: user.role,
      is_impersonating: true,
    };

    return {
      access_token: this.jwtService.sign(payload),
    };
  }
}

import { Injectable, CanActivate, ExecutionContext } from '@nestjs/common';
import { Reflector } from '@nestjs/core';
import { Request } from 'express';

@Injectable()
export class RolesGuard implements CanActivate {
  constructor(private reflector: Reflector) {}

  canActivate(context: ExecutionContext): boolean {
    const requiredRoles = this.reflector.get<string[]>(
      'roles',
      context.getHandler(),
    );

    if (!requiredRoles) {
      return true;
    }

    const request = context
      .switchToHttp()
      .getRequest<Request & { user?: { role: string } }>();
    const user = request.user;
    if (!user) return false;

    return requiredRoles.includes(user.role);
  }
}

import {
  Injectable,
  NestInterceptor,
  ExecutionContext,
  CallHandler,
} from '@nestjs/common';
import { Observable } from 'rxjs';
import { map } from 'rxjs/operators';
import { AuthUserPayload } from '../auth/user.decorator';
import { Request } from 'express';

@Injectable()
export class ImpersonationInterceptor implements NestInterceptor {
  intercept(context: ExecutionContext, next: CallHandler): Observable<unknown> {
    const request = context
      .switchToHttp()
      .getRequest<Request & { user?: AuthUserPayload }>();
    const user = request.user;

    return next.handle().pipe(
      map((response: { data?: any[] } | undefined) => {
        if (!user?.is_impersonating || !response?.data) {
          return response;
        }

        const maskedData = response.data.map(
          (inv: {
            client_name: string;
            amount: number;
            [key: string]: unknown;
          }) => {
            return {
              ...inv,
              client_name: '***',
              amount: 0,
            };
          },
        );

        return { ...response, data: maskedData };
      }),
    );
  }
}

  constructor(
    private readonly invoicesService: InvoicesService,
    private readonly pdfService: PdfService,
  ) {}

  @UseGuards(AuthGuard)
  @Post()
  async create(
    @AuthUser() user: AuthUserPayload,
    @Body() body: CreateInvoiceDto,
  ) {
    const userId = user.sub;

    return this.invoicesService.createInvoice(
      userId,
      body.client_name,
      body.amount,
    );
  }

  @UseGuards(AuthGuard)
  @UseInterceptors(ImpersonationInterceptor)
  @Get()
  async getAll(@AuthUser() user: AuthUserPayload) {
    const userId = user.sub;
    return this.invoicesService.getInvoices(userId);
  }

  @UseGuards(AuthGuard)
  @Get(':id/pdf')
  async downloadPdf(
    @AuthUser() user: AuthUserPayload,
    @Param('id') id: string,
  ) {
    if (user.is_impersonating) {
      throw new ForbiddenException(
        'This action is not permitted in admin mode.',
      );
    }

    const userId = user.sub;
    const invoice = await this.invoicesService.getInvoiceById(userId, id);

    if (invoice.pdf_url) {
      return { url: invoice.pdf_url };
    }

    const pdfUrl = await this.pdfService.generateInvoicePdf(
      invoice.client_name,
      Number(invoice.amount),
      invoice.is_pro,
    );

    await this.invoicesService.savePdfUrl(id, pdfUrl);

    return { url: pdfUrl };
  }

  @UseGuards(AuthGuard, RolesGuard)
  @Roles('OWNER')
  @Post('upgrade')
  async upgradeToPro(@AuthUser() user: AuthUserPayload) {
    const userId = user.sub;
    return this.invoicesService.upgradeWorkspace(userId);
  }

  @UseGuards(AuthGuard, RolesGuard)

import {
  Inject,
  Injectable,
  NotFoundException,
  ForbiddenException,
} from '@nestjs/common';
import { Kysely } from 'kysely';
import { Database } from '../database/database';

@Injectable()
export class InvoicesService {
  constructor(@Inject('DB_INSTANCE') private readonly db: Kysely<Database>) {}

  async createInvoice(userId: string, clientName: string, amount: number) {
    const workspace = await this.db
      .selectFrom('workspaces')
      .select(['id', 'is_pro'])
      .where('user_id', '=', userId)
      .executeTakeFirst();

    if (!workspace) {
      throw new NotFoundException('Workspace not found');
    }

    if (!workspace.is_pro) {
      const { count } = await this.db
        .selectFrom('invoices')
        .select((eb) => eb.fn.count('id').as('count'))
        .where('workspace_id', '=', workspace.id)
        .executeTakeFirstOrThrow();

      const currentInvoicesCount = Number(count);

      if (currentInvoicesCount >= 3) {
        throw new ForbiddenException(
          'The limit on free invoices has been reached',
        );
      }
    }
    const newInvoice = await this.db
      .insertInto('invoices')
      .values({
        workspace_id: workspace.id,
        client_name: clientName,
        amount: amount,
        status: 'PENDING',
      })
      .returningAll()
      .executeTakeFirstOrThrow();

    return newInvoice;
  }

  async getInvoices(userId: string) {
    const workspace = await this.db
      .selectFrom('workspaces')
      .select(['id', 'is_pro'])
      .where('user_id', '=', userId)
      .executeTakeFirst();

    if (!workspace) {
      return { data: [], isPro: false };
    }

    const invoices = await this.db
      .selectFrom('invoices')
      .selectAll()
      .where('workspace_id', '=', workspace.id)
      .execute();

    return {
      data: invoices,
      isPro: workspace.is_pro,
    };
  }

  async getInvoiceById(userId: string, invoiceId: string) {
    const workspace = await this.db
      .selectFrom('workspaces')
      .select(['id', 'is_pro'])
      .where('user_id', '=', userId)
      .executeTakeFirst();

    if (!workspace) {
      throw new NotFoundException('Workspace not found');
    }

    const invoices = await this.db
      .selectFrom('invoices')
      .selectAll()
      .where('id', '=', invoiceId)
      .where('workspace_id', '=', workspace.id)
      .executeTakeFirst();

    if (!invoices) {
      throw new NotFoundException('Invoice not found');
    }
    return { ...invoices, is_pro: workspace.is_pro };
  }

  async upgradeWorkspace(userId: string) {
    await this.db
      .updateTable('workspaces')
      .set({ is_pro: true })
      .where('user_id', '=', userId)
      .execute();

    return { success: true, message: 'Updated to Pro!' };
  }

  async savePdfUrl(invoiceId: string, pdfUrl: string) {
    await this.db
      .updateTable('invoices')
      .set({ pdf_url: pdfUrl })
      .where('id', '=', invoiceId)
      .execute();
  }

  async getAllWorkspacesForAdmin() {
    const workspaces = await this.db
      .selectFrom('workspaces')
      .innerJoin('users', 'users.id', 'workspaces.user_id')
      .select([
        'workspaces.id as workspace_id',
        'workspaces.name as workspace_name',
        'workspaces.is_pro',
        'users.email as owner_email',
        'users.id as owner_id',
      ])
      .execute();

    return workspaces;
  }

  async deleteInvoice(userId: string, invoiceId: string) {
    const workspace = await this.db
      .selectFrom('workspaces')
      .select(['id'])
      .where('user_id', '=', userId)
      .executeTakeFirst();

    if (!workspace) {
      throw new NotFoundException('Workspace not found');
    }

    await this.db
      .deleteFrom('invoices')
      .where('id', '=', invoiceId)
      .where('workspace_id', '=', workspace.id)
      .execute();

    return {
      success: true,
      message: 'The invoice has been successfully deleted',
    };
  }

  async updateInvoice(
    userId: string,
    invoiceId: string,
    clientName: string,
    amount: number,
  ) {
    const workspace = await this.db
      .selectFrom('workspaces')
      .select(['id'])
      .where('user_id', '=', userId)
      .executeTakeFirst();

    if (!workspace) {
      throw new NotFoundException('Workspace not found');
    }

    await this.db
      .updateTable('invoices')
      .set({
        client_name: clientName,
        amount: amount,
        pdf_url: null,
      })
      .where('id', '=', invoiceId)
      .where('workspace_id', '=', workspace.id)
      .execute();

    return {
      success: true,
      message: 'The invoice has been successfully updated',
    };
  }
}

import { S3Client, PutObjectCommand } from '@aws-sdk/client-s3';
import { Injectable } from '@nestjs/common';
import PDFDocument from 'pdfkit';

@Injectable()
export class PdfService {
  private s3Client = new S3Client({
    region: 'auto',
    endpoint: process.env.R2_ENDPOINT,
    credentials: {
      accessKeyId: process.env.R2_ACCESS_KEY_ID!,
      secretAccessKey: process.env.R2_SECRET_ACCESS_KEY!,
    },
  });

  async generateInvoicePdf(clientName: string, amount: number, isPro: boolean) {
    const doc = new PDFDocument({ margin: 50 });

    doc.fontSize(24).text('INVOICE', { align: 'center' }).moveDown(2);
    doc.fontSize(16).text(`Client: ${clientName}`).moveDown();

    doc
      .fontSize(16)
      .text(`Total Amount: $${(amount / 100).toFixed(2)}`)
      .moveDown(3);

    doc
      .fontSize(12)
      .font('Helvetica-Oblique')
      .text('Thank you for your business!');

    if (!isPro) {
      doc.save();
      doc
        .fontSize(40)
        .fillColor('gray')
        .opacity(0.2)
        .rotate(-30, { origin: [doc.page.width / 2, doc.page.height / 2] })
        .text('Created in MicroInvoicer', 0, doc.page.height / 2, {
          align: 'center',
          width: doc.page.width,
        });
      doc.restore();
    }

    const pdfBuffer = await new Promise<Buffer>((resolve, reject) => {
      const chunks: Uint8Array[] = [];
      doc.on('data', (chunk: Uint8Array) => chunks.push(chunk));
      doc.on('end', () => resolve(Buffer.concat(chunks)));
      doc.on('error', reject);
      doc.end();
    });
    const fileName = `invoice-${Date.now()}.pdf`;
    const command = new PutObjectCommand({
      Bucket: process.env.R2_BUCKET_NAME,
      Key: fileName,
      Body: pdfBuffer,
      ContentType: 'application/pdf',
    });
    await this.s3Client.send(command);

    return `${process.env.R2_PUBLIC_URL}/${fileName}`;
  }
}

import { useEffect, useState } from "react";
import Image from "next/image";
import { useRouter } from "next/navigation";
import { InvoiceDTO } from "@micro-invoicer/shared";
import toast from "react-hot-toast";

export default function Home() {
  const [isLoading, setIsLoading] = useState(true);
  const [isServerDown, setIsServerDown] = useState(false);
  const [invoices, setInvoices] = useState<InvoiceDTO[]>([]);
  const [downloadingId, setDownloadingId] = useState<string | null>(null);
  const [isPro, setIsPro] = useState(false);
  const [role, setRole] = useState<string | null>(null);
  const [isAdminImpersonating, setIsAdminImpersonating] = useState(false);
  const [invoiceToDelete, setInvoiceToDelete] = useState<string | null>(null);
  const [showLogoutConfirm, setShowLogoutConfirm] = useState(false);
  const router = useRouter();

  useEffect(() => {
    const getToken = localStorage.getItem("token") as string;

    if (!getToken) {
      router.push("/login");
      return;
    }

    async function fetchInvoices() {
      try {
        const payload = JSON.parse(atob(getToken.split(".")[1]));
        setRole(payload.role);

        if (localStorage.getItem("admin_token")) {
          setIsAdminImpersonating(true);
        }

        const response = await fetch("http://localhost:4000/invoices", {
          method: "GET",
          headers: { Authorization: `Bearer ${getToken}` },
        });

        if (response.ok) {
          const responseData = await response.json();
          setInvoices(responseData.data);
          setIsPro(responseData.isPro);
          setIsServerDown(false);
        } else {
          localStorage.removeItem("token");
          router.push("/login");
        }
      } catch {
        setIsServerDown(true);
        toast.error("Сервер перезапускається...", { id: "server-boot" });
        setTimeout(() => {
          fetchInvoices();
        }, 3000);
      } finally {
        setIsLoading(false);
      }
    }

    fetchInvoices();
  }, [router]);

  async function handleDownload(invoiceId: string) {
    setDownloadingId(invoiceId);
    const getToken = localStorage.getItem("token");
    try {
      if (!getToken) {
        toast.error("Сесія застаріла, введіть логін повторно");
        router.push("/login");
        return;
      } else {
        const response = await fetch(
          `http://localhost:4000/invoices/${invoiceId}/pdf`,
          {
            method: "GET",
            headers: { Authorization: `Bearer ${getToken}` },
          },
        );
        if (response.ok) {
          const data = await response.json();
          window.open(data.url, "_blank");
          toast.success("Інвойс відкрито!");
        } else {
          if (response.status === 401) {
            toast.error("Сесія застаріла, введіть логін повторно");
            localStorage.removeItem("token");
            router.push("/login");
          } else {
            const errorData = await response.json().catch(() => ({}));
            toast.error(errorData?.message || "Дія заборонена.");
          }
        }
      }
    } catch {
      toast.error("Помилка з'єднання з сервером");
    } finally {
      setDownloadingId(null);
    }
  }

  function confirmDelete(id: string) {
    setInvoiceToDelete(id);
  }

  async function executeDelete() {
    if (!invoiceToDelete) return;

    const getToken = localStorage.getItem("token");
    try {
      if (!getToken) {
        toast.error("Сесія застаріла, введіть логін повторно");
        router.push("/login");
        return;
      }

      const response = await fetch(
        `http://localhost:4000/invoices/${invoiceToDelete}`,
        {
          method: "DELETE",
          headers: { Authorization: `Bearer ${getToken}` },
        },
      );

      if (response.ok) {
        toast.success("Інвойс видалено!");
        setInvoices((prev) => prev.filter((inv) => inv.id !== invoiceToDelete));
      } else {
        toast.error("Не вдалося видалити інвойс.");
      }
    } catch {
      toast.error("Помилка з'єднання з сервером");
    } finally {
      setInvoiceToDelete(null);
    }
  }

  const handleLogout = () => {
    localStorage.removeItem("token");
    router.push("/login");
    toast.success("Ви вийшли з аккаунту!");
  };

  const handleLeaveImpersonation = () => {
    const adminToken = localStorage.getItem("admin_token");
    if (adminToken) {
      localStorage.setItem("token", adminToken);
      localStorage.removeItem("admin_token");
      toast.success("Ви повернулися в адмінку!");
      router.push("/admin");
    }
  };

  return (
    <div>
      {isServerDown && (
        <div className="fixed inset-0 z-50 flex flex-col items-center justify-center bg-gray-50/90 backdrop-blur-sm">
          <svg
            className="mb-4 h-12 w-12 animate-spin text-zinc-900"
            xmlns="http://www.w3.org/2000/svg"
            fill="none"
            viewBox="0 0 24 24"
          >
            <circle
              className="opacity-25"
              cx="12"
              cy="12"
              r="10"
              stroke="currentColor"
              strokeWidth="4"
            ></circle>
            <path
              className="opacity-75"
              fill="currentColor"
              d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"
            ></path>
          </svg>
          <h2 className="text-2xl font-bold text-gray-900">
            Перезапускаємо сервер...
          </h2>
          <p className="mt-2 text-gray-500">
            Це може зайняти кілька секунд. Будь ласка, зачекайте.
          </p>
        </div>
      )}
      <div className="min-h-screen bg-gray-50 p-8 font-sans">
        {isAdminImpersonating && (
          <div className="mx-auto mb-8 max-w-4xl">
            <div className="flex items-center justify-between rounded-xl bg-zinc-800 px-6 py-4 shadow-lg ring-1 ring-slate-800">

      try {
        const response = await fetch(
          `http://localhost:4000/invoices/${params?.id}`,
          {
            method: "GET",
            headers: {
              "Content-Type": "application/json",
              Authorization: `Bearer ${getToken}`,
            },
          },
        );

        if (response.ok) {
          const data = await response.json();
          reset({ clientName: data.client_name, amount: data.amount / 100 });
        } else {
          if (response.status === 401) {
            localStorage.removeItem("token");
            router.push("/login");
            return;
          }
          const errorData = await response.json().catch(() => ({}));
          toast.error(
            errorData?.message || "Не вдалося завантажити дані інвойсу.",
          );
          router.push("/");
        }
      } catch {
        toast.error("Помилка з'єднання з сервером.");
      }
    }

    if (params?.id) {
      fetchInvoice();
    }
  }, [params?.id, reset, router]);

  async function onSubmit(data: InvoiceFormOutput) {
    setIsSubmitting(true);
    try {
      const getToken = localStorage.getItem("token");

      if (!getToken) {
        toast.error("Сесія застаріла, введіть логін повторно");
        router.push("/login");
        return;
      } else {
        const response = await fetch(
          `http://localhost:4000/invoices/${params.id}`,
          {
            method: "PATCH",
            headers: {
              "Content-Type": "application/json",
              Authorization: `Bearer ${getToken}`,
            },
            body: JSON.stringify({
              client_name: data.clientName,
              amount: Math.round(data.amount * 100),
            }),
          },
        );

        if (!response.ok) {
          if (response.status === 403) {
            router.push("/pro");
          }
          toast.error("Не вдалося оновити інвойс..");
          return;
        } else {
          toast.success("Дані успішно оновлені!");
          router.push("/");
        }
      }
    } catch {
      toast.error("Помилка з'єднання з сервером");
    } finally {
      setIsSubmitting(false);
    }
  }

  return (
    <div className="flex min-h-screen items-center justify-center bg-gray-50 px-4 font-sans">
      <form
        onSubmit={handleSubmit(onSubmit)}
        className="w-full max-w-md space-y-6 rounded-2xl bg-white p-10 shadow-xl ring-1 ring-gray-900/5"
      >
        <h2 className="mb-8 text-center text-3xl font-extrabold tracking-tight text-gray-900">
          Редагувати інвойс
        </h2>

        <div className="space-y-4">
          <div>
            <input
              {...register("clientName")}
              className="block w-full rounded-xl border border-gray-300 px-4 py-3 text-gray-900 placeholder-gray-400 transition-all focus:border-zinc-900 focus:outline-none focus:ring-2 focus:ring-zinc-900/20 sm:text-sm"
              placeholder="Ім'я клієнта (англійською)"
            />
            {errors.clientName && (
              <p className="mt-1 text-sm text-red-500">
                {errors.clientName.message}
              </p>
            )}
          </div>

          <div>
            <input
              type="number"
              {...register("amount")}
              className="block w-full rounded-xl border border-gray-300 px-4 py-3 text-gray-900 placeholder-gray-400 transition-all focus:border-zinc-900 focus:outline-none focus:ring-2 focus:ring-zinc-900/20 sm:text-sm"
              placeholder="Сума ($)"
            />
            {errors.amount && (
              <p className="mt-1 text-sm text-red-500">
                {errors.amount.message}
              </p>
            )}
          </div>
        </div>

        <div className="flex gap-3 pt-2">
          <button
            type="button"
            onClick={() => router.push("/")}
            className="flex w-full justify-center rounded-xl border border-gray-200 bg-white px-4 py-3 text-sm font-bold text-gray-700 shadow-sm transition-all hover:bg-gray-50 focus:outline-none focus:ring-2 focus:ring-gray-200 focus:ring-offset-2"
          >
            Скасувати
          </button>
          <button
            disabled={isSubmitting}
            type="submit"
            className="flex w-full justify-center rounded-xl bg-zinc-900 px-4 py-3 text-sm font-bold text-white shadow-sm transition-all hover:-translate-y-0.5 hover:bg-zinc-800 focus:outline-none focus:ring-2 focus:ring-zinc-900 focus:ring-offset-2"
          >
            {isSubmitting ? "Збереження..." : "Зберегти"}
          </button>
        </div>
      </form>
    </div>
  );
}
